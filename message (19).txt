-- [ЛОГИКА РАБОТЫ С ЧЕРНЫМ СПИСКОМ]
local function getSharedBlacklist() if isfile(FILE_NAME) then local ok, res = pcall(function() return Http:JSONDecode(readfile(FILE_NAME)) end); if ok then return res end end; return {} end
local function addToBlacklist(jobId) local list = getSharedBlacklist(); if not table.find(list, jobId) then table.insert(list, jobId); writefile(FILE_NAME, Http:JSONEncode(list)) end end

-- 🔥 [ГЛАВНЫЙ ЦИКЛ, КОТОРЫЙ РЕШАЕТ ВСЁ]
print("CapyFinder запущен с исправленным циклом.")
if not game:IsLoaded() then game.Loaded:Wait() end
if not LocalPlayer.Character then LocalPlayer.CharacterAdded:Wait() end

local server_cursor = ""
while true do
    print("\n--- НАЧАЛО НОВОГО ЦИКЛА ---")
    local current_job_id = game.JobId
    
    -- 1. Добавляем текущий сервер в черный список, чтобы не зайти на него снова
    addToBlacklist(current_job_id)
    
    -- 2. Сканируем текущий сервер
    local foundItems = aggressiveScan()
    
    -- 3. Обрабатываем находки
    if #foundItems > 0 then
        print("✅ НАЙДЕНО! Отправляю лог...")
        table.sort(foundItems, function(a,b) return a.gen > b.gen end)
        local best = foundItems[1]
        if best.gen >= HIGH_GEN_MIN and not checkBlacklist(best.name) then
            notify(foundItems, "High", 0xFF0000)
        else
            notify(foundItems, "Mid", 0xFFA500)
        end
    else 
        print("❌ Ничего не найдено.")
        task.spawn(notifyNotFound)
    end
    
    -- 4. Ищем следующий сервер для прыжка
    print("🔎 Ищу следующий сервер...")
    local targetFound = false
    while not targetFound do
        local url = "https://games.roblox.com/v1/games/" .. PLACE_ID .. "/servers/Public?sortOrder=Desc&limit=100&excludeFullGames=true&_nocache=" .. math.random(1000000)
        if server_cursor ~= "" then url = url .. "&cursor=" .. server_cursor end
        
        local success, response = pcall(function() return Http:JSONDecode(game:HttpGet(url)) end)
        
        if success and response and response.data then
            local servers = response.data
            server_cursor = response.nextPageCursor or ""
            
            -- Перемешиваем сервера, чтобы боты не прыгали на одни и те же
            for i = #servers, 2, -1 do local j = math.random(i); servers[i], servers[j] = servers[j], servers[i] end
            
            local currentBlacklist = getSharedBlacklist()
            for _, server in ipairs(servers) do
                if server.id ~= current_job_id and server.playing < server.maxPlayers and not table.find(currentBlacklist, server.id) then
                    print("🚀 Прыгаю на сервер: " .. server.id)
                    targetFound = true
                    TPS:TeleportToPlaceInstance(PLACE_ID, server.id, LocalPlayer)
                    break 
                end
            end
        else
            -- Если HttpGet не удался, ждем и пробуем снова
            task.wait(2)
        end
        
        if not targetFound then
            if server_cursor == "" then
                print("⚠️ Сервера закончились. Начинаю сначала или прыгаю на случайный.")
                server_cursor = "" -- Сбрасываем курсор
                TPS:Teleport(PLACE_ID, LocalPlayer) -- Прыгаем на случайный публичный сервер
                targetFound = true -- Выходим из цикла поиска, так как телепорт уже инициирован
            end
        end
    end
    
    -- 5. Ждем после инициации телепорта
    print("⏳ Ожидание после телепорта: " .. HOPPER_WAIT .. "с")
    task.wait(HOPPER_WAIT)
end
