-- ⚡ HYPER FAST SERVER HOPPER (GUI + CACHE)
-- Кэширует сервера для скорости + Кнопка Стоп

local Http = game:GetService("HttpService")
local TPS = game:GetService("TeleportService")
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local CoreGui = game:GetService("CoreGui")

local PlaceId = game.PlaceId
local HOPPING = false
local CACHED_SERVERS = {}
local Cursor = ""

-- // 1. СОЗДАНИЕ GUI //

Screen.Name = "HyperHopGUI"
Screen.Parent = CoreGui

local Frame = Instance.new("Frame", Screen)

-- // 2. ЛОГИКА СЕРВЕРОВ //
local function UpdateStatus(txt) StatusLbl.Text = txt end

local function FetchServers()
    UpdateStatus("♻️ Обновление списка...")
    local API = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Desc&limit=100&excludeFullGames=true"
    if Cursor ~= "" then API = API .. "&cursor=" .. Cursor end
    
    local s, r = pcall(game.HttpGet, game, API)
    if not s then return end
    
    local data = Http:JSONDecode(r)
    if not data or not data.data then return end
    
    -- Сохраняем курсор для следующей страницы
    Cursor = data.nextPageCursor or ""
    
    -- Очищаем кэш и заливаем новые
    CACHED_SERVERS = {}
    for _, s in ipairs(data.data) do
        if s.playing < s.maxPlayers and s.id ~= game.JobId then
            table.insert(CACHED_SERVERS, s.id)
        end
    end
    UpdateStatus("✅ Кэшировано: " .. #CACHED_SERVERS)
end

local function Teleport(id)
    UpdateStatus("🚀 Вход: " .. id:sub(1,8))
    TPS:TeleportToPlaceInstance(PlaceId, id, Players.LocalPlayer)
end

-- // 3. МГНОВЕННЫЙ ПЕРЕЗАПУСК ПРИ ОШИБКЕ //
TPS.TeleportInitFailed:Connect(function(player, result, errorMessage)
    if player == Players.LocalPlayer then
        UpdateStatus("❌ Ошибка ("..tostring(result)..")")
        -- Не ждем таймер, сразу ищем новый сервер в цикле
    end
end)

-- // 4. ГЛАВНЫЙ ЦИКЛ //
task.spawn(function()
    while true do
        if HOPPING then
            -- Если сервера кончились, качаем пачку
            if #CACHED_SERVERS == 0 then
                FetchServers()
                task.wait(0.5)
            else
                -- Берем случайный из кэша (СУПЕР БЫСТРО)
                local idx = math.random(1, #CACHED_SERVERS)
                local target = CACHED_SERVERS[idx]
                
                -- Удаляем из списка, чтобы не повторяться
                table.remove(CACHED_SERVERS, idx)
                
                if target then
                    Teleport(target)
                    
                    -- Ждем совсем чуть-чуть. Если мы все еще тут - цикл продолжится
                    -- Если телепорт начнется - скрипт сам заморозится игрой
                    local t = tick()
                    while tick() - t < 2 do -- Ждем максимум 2 сек, раньше было вечность
                        if not HOPPING then break end
                        RunService.Heartbeat:Wait()
                    end
                end
            end
        else
            UpdateStatus("💤 Остановлено")
            task.wait(1)
        end
    end
end)

-- // 5. КНОПКА //
ToggleBtn.MouseButton1Click:Connect(function()
    HOPPING = not HOPPING
    if HOPPING then
        ToggleBtn.Text = "STOP HOPPING"
        ToggleBtn.BackgroundColor3 = Color3.fromRGB(200, 40, 40)
    else
        ToggleBtn.Text = "START HOPPING"
        ToggleBtn.BackgroundColor3 = Color3.fromRGB(40, 200, 40)
    end
end)
