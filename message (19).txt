-- ⚡ FLASH SERVER HOPPER
-- Быстрый. Рандомный. Работает.

local Http = game:GetService("HttpService")
local TPS = game:GetService("TeleportService")
local Players = game:GetService("Players")

local PlaceId = game.PlaceId
local Cursor = ""

local function Hop()
    while true do
        -- 1. Запрос (Сразу просим убрать полные сервера для скорости)
        local API = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Desc&limit=100&excludeFullGames=true"
        if Cursor ~= "" then API = API .. "&cursor=" .. Cursor end
        
        local success, response = pcall(game.HttpGet, game, API)
        
        if not success then
            task.wait(1) -- Если ошибка сети, ждем секунду и пробуем снова
            continue
        end
        
        local data = Http:JSONDecode(response)
        if not data or not data.data then 
            task.wait(1) 
            continue 
        end
        
        -- 2. Ищем подходящий сервер
        local servers = {}
        for _, s in ipairs(data.data) do
            -- Проверка: не текущий сервер, и есть место (хотя API уже должен отфильтровать)
            if s.playing < s.maxPlayers and s.id ~= game.JobId then
                table.insert(servers, s.id)
            end
        end
        
        -- 3. Телепортация (Рандомная)
        if #servers > 0 then
            local target = servers[math.random(1, #servers)] -- Случайный выбор
            
            -- Попытка телепорта
            TPS:TeleportToPlaceInstance(PlaceId, target, Players.LocalPlayer)
            
            -- Ждем немного, если не улетели - цикл пойдет дальше
            task.wait(2)
        end
        
        -- 4. Листаем страницы, если на этой ничего не нашли
        if data.nextPageCursor then
            Cursor = data.nextPageCursor
        else
            Cursor = "" -- Начинаем сначала
        end
    end
end

-- GUI статус (чтобы видеть, что он работает)
local sg = Instance.new("ScreenGui", game.CoreGui)
local tl = Instance.new("TextLabel", sg)
tl.Size = UDim2.new(0,200,0,50); tl.Position = UDim2.new(0.5,-100,0,0)
tl.BackgroundColor3 = Color3.new(0,0,0); tl.TextColor3 = Color3.new(1,1,1)
tl.Text = "⚡ Hopping..."

-- Запуск
Hop()
