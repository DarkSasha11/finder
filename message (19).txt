local function getBrainrotImage(name) 
    if not name or name == "" then return nil end 
    return "https://steal-a-brainrot.org/_next/image?url=%2Fimages%2Fbrainrots%2F" .. string.lower(name):gsub(" ", "-") .. ".webp&w=1920&q=90" 
end

local function shortMoney(v) 
    v = tonumber(v) or 0
    if v >= 1e9 then 
        local f = string.format("%.2f", v/1e9):gsub("%.?0+$", "")
        return "$" .. f .. "B/s" 
    elseif v >= 1e6 then 
        local f = string.format("%.2f", v/1e6):gsub("%.?0+$", "")
        return "$" .. f .. "M/s" 
    elseif v >= 1e3 then 
        return string.format("$%.0fK/s", v/1e3) 
    else 
        return string.format("$%d/s", math.floor(v)) 
    end 
end

local function stripRichText(s) 
    s = type(s) == "string" and s or ""
    s = s:gsub("<.->", ""):gsub("%s+", " "):gsub("^%s+", ""):gsub("%s+$", "")
    return s 
end

local function updateJsonBlob(newLogData)
    -- 1. Скачиваем текущие данные (GET)
    local currentLogs = {}
    local get_response = request({Url = API_URL, Method = "GET"})
    
    if get_response and get_response.StatusCode == 200 then
        local success, data = pcall(HttpService.JSONDecode, HttpService, get_response.Body)
        if success and type(data) == "table" then 
            currentLogs = data 
        end
    end
    
    -- 2. Добавляем новый лог в начало
    table.insert(currentLogs, 1, newLogData)
    
    -- 3. Чистим старые логи (оставляем последние 40)
    while #currentLogs > 40 do 
        table.remove(currentLogs) 
    end
    
    -- 4. Отправляем обновленный список обратно (PUT)
    request({
        Url = API_URL, 
        Method = "PUT", 
        Headers = {
            ["Content-Type"] = "application/json", 
            ["Accept"] = "application/json"
        }, 
        Body = HttpService:JSONEncode(currentLogs)
    })
    
    print("✅ [Valkyrie] Данные отправлены в JsonBlob для GUI.")
end

local function isValidWebhook(url) 
    return type(url) == "string" and url:find("discord.com/api/webhooks") 
end

local function pickRangeWebhook(mps) 
    mps = tonumber(mps) or 0
    if mps >= 1000000000 then 
        return "1B+", WEBHOOK_1B_PLUS 
    elseif mps >= 400000000 then 
        return "400M–1B", WEBHOOK_400_1B 
    elseif mps >= 100000000 then 
        return "100–400M", WEBHOOK_100_400M 
    elseif mps >= 50000000 then 
        return "50–100M", WEBHOOK_50_100M 
    elseif mps >= 10000000 then 
        return "10–50M", WEBHOOK_10_50M 
    elseif mps >= 1000000 then 
        return "1–10M", WEBHOOK_1_10M 
    else 
        return nil, nil 
    end 
end

local ReplicatedStorage, Packages, Datas, Shared, Utils
local Synchronizer, AnimalsData, AnimalsShared, NumberUtils

local function waitForModules()
    pcall(function() ReplicatedStorage = game:GetService("ReplicatedStorage") end)
    if not ReplicatedStorage then return false end
    
    local attempts = 0
    while attempts < 30 do
        local p1,p2,p3,p4 = pcall(function() 
            Packages = ReplicatedStorage:WaitForChild("Packages")
            Datas = ReplicatedStorage:WaitForChild("Datas")
            Shared = ReplicatedStorage:WaitForChild("Shared")
            Utils = ReplicatedStorage:WaitForChild("Utils")
        end)
        
        if p1 then 
            local s1,m1 = pcall(require, Packages:WaitForChild("Synchronizer"))
            local s2,m2 = pcall(require, Datas:WaitForChild("Animals"))
            local s3,m3 = pcall(require, Shared:WaitForChild("Animals"))
            local s4,m4 = pcall(require, Utils:WaitForChild("NumberUtils"))
            
            if s1 and s2 and s3 and s4 then 
                Synchronizer = m1
                AnimalsData = m2
                AnimalsShared = m3
                NumberUtils = m4
                return true
            end 
        end
        attempts = attempts + 1
        task.wait(1)
    end
    return false
end

local function scanBrainrots()
    local found = {}
    local plots = workspace:FindFirstChild("Plots")
    if not plots or not Synchronizer or not AnimalsData or not AnimalsShared then return {} end

    for _, plot in ipairs(plots:GetChildren()) do
        local ch = Synchronizer:Get(plot.Name)
        if not ch then continue end
        local list = ch:Get("AnimalList")
        if typeof(list) ~= "table" then continue end

        for _, animal in pairs(list) do
            if animal and animal.Index then
                local info = AnimalsData[animal.Index]
                if info then
                    local mps = AnimalsShared:GetGeneration(animal.Index, animal.Mutation, animal.Traits)
                    if mps and mps >= RANGE2_MIN_MPS then
                        table.insert(found, {name = info.DisplayName or animal.Index, mps = mps})
                    end
                end
            end
        end
    end

    if #found > 0 then
        local seen = {}
        local uniqueFound = {}
        for _, item in ipairs(found) do
            local key = item.name .. "|" .. tostring(math.floor(item.mps))
            if not seen[key] then
                seen[key] = true
                table.insert(uniqueFound, item)
            end
        end
        
        table.sort(uniqueFound, function(a, b) return a.mps > b.mps end)
        return uniqueFound
    end
    return {}
end
