-- ⚡ ULTRA FAST SERVER HOPPER (FIXED 771 ERROR)
local TeleportService = game:GetService("TeleportService")
local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")
local GuiService = game:GetService("GuiService")
local RunService = game:GetService("RunService")

local SETTINGS = {
    GAME_ID = 109983668079237,
    URL = "https://raw.githubusercontent.com/velo35001/logi/refs/heads/main/logi.txt",
    TIMEOUT = 6, -- Если через 6 сек мы еще тут - пробуем следующий
}

local BLACKLIST = {}
local SERVERS = {}
local ATTEMPTING = false

-- // GUI //
local screen = Instance.new("ScreenGui", game.CoreGui)
local frame = Instance.new("Frame", screen)
frame.Size = UDim2.new(0, 200, 0, 70)
frame.Position = UDim2.new(0.5, -100, 0, 10)
frame.BackgroundColor3 = Color3.new(0.1, 0.1, 0.1)
local label = Instance.new("TextLabel", frame)
label.Size = UDim2.new(1,0,1,0)
label.BackgroundTransparency = 1
label.TextColor3 = Color3.new(1,1,1)
label.TextWrapped = true
label.Font = Enum.Font.GothamBold
label.TextSize = 14

local function Status(txt) label.Text = txt end

-- // ФУНКЦИЯ ПОЛУЧЕНИЯ СЕРВЕРОВ //
local function GetServers()
    Status("🔄 Скачиваю список...")
    local s, r = pcall(game.HttpGet, game, SETTINGS.URL)
    if not s then return {} end
    
    local t = {}
    for id in r:gmatch("%w+-%w+-%w+-%w+-%w+") do
        table.insert(t, id)
    end
    return t
end

-- // ФУНКЦИЯ ОЧИСТКИ ОШИБОК //
local function ClearErrors()
    -- Пытаемся закрыть серые окна роблокса
    pcall(function() GuiService:ClearError() end) 
end

-- // ГЛАВНАЯ ЛОГИКА //
local function Hop()
    if ATTEMPTING then return end
    ATTEMPTING = true
    
    if #SERVERS == 0 then
        SERVERS = GetServers()
    end
    
    -- Выбираем сервер
    local targetJobId = nil
    for i = 1, 50 do -- Пробуем найти не забаненный ID
        local rnd = SERVERS[math.random(1, #SERVERS)]
        if rnd and not BLACKLIST[rnd] then
            targetJobId = rnd
            break
        end
    end
    
    if not targetJobId then
        Status("♻️ Список кончился, обновляю...")
        SERVERS = {}
        ATTEMPTING = false
        return
    end

    Status("🚀 Вход в: " .. targetJobId:sub(1,8))
    
    -- Попытка телепорта
    TeleportService:TeleportToPlaceInstance(SETTINGS.GAME_ID, targetJobId, Players.LocalPlayer)
    
    -- ЗАЩИТА ОТ ЗАВИСАНИЯ (FIX 771)
    local start = tick()
    while tick() - start < SETTINGS.TIMEOUT do
        RunService.Heartbeat:Wait()
        -- Если вдруг сработало событие ошибки
    end
    
    -- Если мы всё еще тут через 6 секунд, значит ошибка 771 или Server Full
    Status("⚠️ Тайм-аут! След. сервер...")
    BLACKLIST[targetJobId] = true
    ClearErrors() -- Пробуем закрыть окно ошибки
    ATTEMPTING = false -- Сбрасываем флаг, цикл начнется заново
end

-- // СЛУШАТЕЛИ ОШИБОК //
TeleportService.TeleportInitFailed:Connect(function(pl, res, err)
    Status("❌ Ошибка: " .. tostring(res))
    if ATTEMPTING then
        -- Мгновенно пробуем следующий, не ждем таймера
        ATTEMPTING = false 
        ClearErrors()
    end
end)

-- Запуск цикла
task.spawn(function()
    while true do
        if not ATTEMPTING then
            Hop()
        end
        task.wait(0.1)
    end
end)
