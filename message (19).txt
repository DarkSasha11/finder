-- Перетаскивание
local dragging, dragInput, dragStart, startPos
frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = frame.Position
    end
end)
frame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then dragInput = input end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
end)

local function SetStatus(text, color)
    status.Text = text
    if color then status.TextColor3 = color end
end

-- // ЛОГИКА ТЕЛЕПОРТА //

-- 1. Загрузка списка
local function FetchServers()
    local success, result = pcall(function()
        return game:HttpGet(SETTINGS.PASTEFY_URL, true) -- true отключает кэш
    end)
    
    if not success then
        SetStatus("Ошибка получения списка!", Color3.fromRGB(255, 50, 50))
        return {}
    end

    local list = {}
    for id in result:gmatch("([a-f0-9%-]+)") do
        if #id > 10 then -- Простая проверка, что это похоже на JobId
            table.insert(list, id)
        end
    end
    return list
end

-- 2. Обработка ошибки 771 и других сбоев
TeleportService.TeleportInitFailed:Connect(function(player, result, errorMessage)
    if player == Players.LocalPlayer then
        -- Если телепорт не удался (сервер оффлайн/ошибка 771)
        SetStatus("⛔ Сбой: " .. tostring(result) .. "\nПробуем следующий...", Color3.fromRGB(255, 100, 100))
        
        if currentTarget then
            BLACKLIST[currentTarget] = os.time() -- В бан этот сервер
        end
        
        isTeleporting = false -- Разрешаем искать новый сервер
    end
end)

-- 3. Основной цикл
task.spawn(function()
    while true do
        if not isTeleporting then
            -- Загружаем список, если пуст
            if #SERVER_LIST == 0 then
                SetStatus("📥 Загрузка списка серверов...", Color3.fromRGB(100, 200, 255))
                SERVER_LIST = FetchServers()
                task.wait(0.5)
            end

            -- Ищем валидный сервер
            local validServer = nil
            
            -- Перемешиваем список для рандома (Fisher-Yates shuffle)
            for i = #SERVER_LIST, 2, -1 do
                local j = math.random(i)
                SERVER_LIST[i], SERVER_LIST[j] = SERVER_LIST[j], SERVER_LIST[i]
            end

            for _, id in ipairs(SERVER_LIST) do
                -- Проверка: нет ли в черном списке
                if not BLACKLIST[id] or (os.time() - BLACKLIST[id] > SETTINGS.BLACKLIST_TIME) then
                    validServer = id
                    break
                end
            end

            if validServer then
                isTeleporting = true
                currentTarget = validServer
                
                SetStatus("🚀 Вход: " .. validServer:sub(1, 8) .. "...", Color3.fromRGB(100, 255, 100))
                
                -- Попытка телепорта
                local success, err = pcall(function()
                    TeleportService:TeleportToPlaceInstance(SETTINGS.GAME_ID, validServer, Players.LocalPlayer)
                end)

                if not success then
                    -- Если ошибка возникла сразу (до начала перехода)
                    SetStatus("⚠️ Ошибка вызова: " .. tostring(err), Color3.fromRGB(255, 200, 50))
                    BLACKLIST[validServer] = os.time()
                    isTeleporting = false
                    task.wait(SETTINGS.RETRY_DELAY)
                else
                    -- Если вызов прошел, ждем. Если вылетит Error 771, сработает TeleportInitFailed выше
                    -- и сбросит isTeleporting в false
                    SetStatus("⏳ Подключение... Жди.", Color3.fromRGB(150, 255, 150))
                    task.wait(4) -- Даем время на реакцию Roblox
                end
            else
                SetStatus("♻️ Все сервера в бане/недоступны.\nОбновление списка...", Color3.fromRGB(255, 200, 0))
                SERVER_LIST = {} -- Очищаем, чтобы скачать заново
                task.wait(SETTINGS.REFRESH_DELAY)
            end
        end
        task.wait(0.1)
    end
end)
